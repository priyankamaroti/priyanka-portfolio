osm_monthly_kpi_rollup.sql PostgreSQL 15 
WITH RankedUserEvents AS (
  SELECT 
    user_id,
    session_id,
    event_timestamp,
    ROW_NUMBER() OVER (
      PARTITION BY user_id 
      ORDER BY event_timestamp DESC
    ) AS session_seq
  FROM telemetry.events
  WHERE event_timestamp >= NOW() - INTERVAL '30 days'
)
SELECT 
  user_id,
  COUNT(session_id) AS active_sessions
FROM RankedUserEvents
GROUP BY 1;

YoY_Revenue_Growth = 
VAR CurrentRev = [Total Net Revenue]
VAR PriorYearRev = 
  CALCULATE(
    [Total Net Revenue],
    SAMEPERIODLASTYEAR('DimDate'[Date])
  )
RETURN
  DIVIDE(CurrentRev - PriorYearRev, PriorYearRev, 0)
